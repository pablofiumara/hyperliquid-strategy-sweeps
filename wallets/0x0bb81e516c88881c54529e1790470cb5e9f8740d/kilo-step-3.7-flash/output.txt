current_position = 0.0
entry_prices = []
entry_sizes = []
trade_open_time = None

for fill in fills:
    sz = float(fill["sz"])
    px = float(fill["px"])
    start_pos = float(fill["startPosition"])
    end_pos = start_pos + sz if fill["side"] == "B" else start_pos - sz
    closed_pnl = float(fill["closedPnl"])
    
    if "Open" in fill["dir"]:
        if current_position == 0:
            trade_open_time = fill["time"]
            entry_prices = []
            entry_sizes = []
        entry_prices.append(px)
        entry_sizes.append(sz)
        current_position = end_pos
    elif "Close" in fill["dir"]:
        if current_position > 0 and closed_pnl != 0:
            avg_entry = sum(p*s for p,s in zip(entry_prices, entry_sizes)) / sum(entry_sizes)
            total_sz = sum(entry_sizes)
            hold_ms = fill["time"] - trade_open_time
            trades.append({
                "entry_px": avg_entry,
                "exit_px": px,
                "sz": total_sz,
                "pnl": closed_pnl,
                "hold_ms": hold_ms,
                "open_time": trade_open_time,
                "close_time": fill["time"],
                "roi": (closed_pnl / (avg_entry * total_sz)) * 100,
                "num_fills": len(entry_prices),
            })
            current_position = 0
            entry_prices = []
            entry_sizes = []
