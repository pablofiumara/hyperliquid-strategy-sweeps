import json
from collections import Counter, defaultdict
from datetime import datetime

with open('/tmp/fills_all.json') as f:
    data = json.load(f)

# Daily trading pattern
daily_trades = Counter()
daily_pnl = defaultdict(float)
for f in data:
    dt = datetime.fromtimestamp(f['time']/1000)
    date = dt.date()
    daily_trades[date] += 1
    p = float(f.get('closedPnl', '0'))
    if p != 0:
        daily_pnl[date] += p

# Most active days
print('=== TOP 10 MOST ACTIVE DAYS ===')
for date, count in daily_trades.most_common(10):
    pnl = daily_pnl.get(date, 0)
    print(f'  {date}: {count} trades, PnL: ${pnl:.2f}')

# Monthly analysis
monthly = defaultdict(lambda: {'fills': 0, 'pnl': 0})
for f in data:
    dt = datetime.fromtimestamp(f['time']/1000)
    month = dt.strftime('%Y-%m')
    monthly[month]['fills'] += 1
    monthly[month]['pnl'] += float(f.get('closedPnl', '0'))

print('\n=== MONTHLY BREAKDOWN ===')
for month in sorted(monthly.keys()):
    m = monthly[month]
    print(f'  {month}: {m["fills"]} fills, PnL: ${m["pnl"]:.2f}')

# PnL distribution - how many trades were profitable
profitable = 0
losing = 0
total_pnl = 0
for f in data:
    p = float(f.get('closedPnl', '0'))
    total_pnl += p
    if p > 0:
        profitable += 1
    elif p < 0:
        losing += 1

print(f'\n=== PnL SUMMARY ===')
print(f'Profitable closed trades: {profitable}')
print(f'Losing closed trades: {losing}')
print(f'Total closed PnL: ${total_pnl:.2f}')
print(f'Win rate (of closed): {profitable/(profitable+losing)*100:.1f}%')
