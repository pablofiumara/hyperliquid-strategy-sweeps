# Fetch user activity (fills)
def fetch_activity(user, max_pages=100):
    all_rows = []
    page = 0
    while page < max_pages:
        payload = {"type": "userActivity", "user": user, "page": page}
        r = requests.post(URL, json=payload, timeout=30)
        data = r.json()
        if isinstance(data, list):
            rows = data
        elif isinstance(data, dict):
            rows = data.get("userActivity", data.get("rows", []))
        else:
            rows = []
        if not rows:
            break
        all_rows.extend(rows)
        if len(rows) < 1000:
            break
        page += 1
        time.sleep(0.2)
    return all_rows

import json
from datetime import datetime, timezone
from collections import Counter, defaultdict

with open("/tmp/fills.json") as f:
    data = json.load(f)

def dt(t):
    return datetime.fromtimestamp(t/1000, tz=timezone.utc)

print("="*60)
print("BASIC STATS")
print("="*60)
print("Total fills:", len(data))
times = [d["time"] for d in data]
print("Date range:", dt(min(times)), "to", dt(max(times)))
print("Days span:", (max(times)-min(times))/1000/86400, "days")

print("\n" + "="*60)
print("COINS")
print("="*60)
coins = Counter(d["coin"] for d in data)
for c, n in coins.most_common():
    coin_data = [d for d in data if d["coin"]==c]
    sides = Counter(d["side"] for d in coin_data)
    dirs = Counter(d["dir"] for d in coin_data)
    total_sz = sum(float(d["sz"]) for d in coin_data)
    total_pnl = sum(float(d["closedPnl"]) for d in coin_data)
    total_fee = sum(float(d["fee"]) for d in coin_data)
    t0 = min(d["time"] for d in coin_data)
    t1 = max(d["time"] for d in coin_data)
    print(f"{c:6s} n={n:4d} B={sides.get('B',0):4d} A={sides.get('A',0):4d} "
          f"sz={total_sz:10.2f} pnl={total_pnl:10.2f} fee={total_fee:.4f} "
          f"span={dt(t0).strftime('%m-%d')} to {dt(t1).strftime('%m-%d')}")

print("\n" + "="*60)
print("DAILY ACTIVITY")
print("="*60)
daily = defaultdict(list)
for d in data:
    day = dt(d["time"]).strftime("%Y-%m-%d")
    daily[day].append(d)
print(f"Active days: {len(daily)}")
for day in sorted(daily.keys()):
    rows = daily[day]
    b = sum(1 for r in rows if r["side"]=="B")
    a = sum(1 for r in rows if r["side"]=="A")
    pnl = sum(float(r["closedPnl"]) for r in rows)
    print(f"{day} n={len(rows):3d} B={b:3d} A={a:3d} pnl={pnl:8.2f}")

print("\n" + "="*60)
print("POSITION SIZING")
print("="*60)
szs = [float(d["sz"]) for d in data]
print(f"Min sz: {min(szs):.4f}  Max sz: {max(szs):.4f}  Mean sz: {sum(szs)/len(szs):.4f}")
# Distribution
import statistics
print(f"Median sz: {statistics.median(szs):.4f}")
# Size buckets
buckets = Counter()
for s in szs:
    if s < 1: buckets["<1"] += 1
    elif s < 5: buckets["1-5"] += 1
    elif s < 20: buckets["5-20"] += 1
    elif s < 100: buckets["20-100"] += 1
    else: buckets[">100"] += 1
print("Size buckets:", dict(buckets))

print("\n" + "="*60)
print("PRICES")
print("="*60)
prices = [float(d["px"]) for d in data]
print(f"Price range: {min(prices):.4f} to {max(prices):.4f}")
