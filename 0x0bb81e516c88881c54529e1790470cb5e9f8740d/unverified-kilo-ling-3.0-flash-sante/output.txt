import json, sys
data = json.load(sys.stdin)
data.sort(key=lambda x: x['time'])

# Detailed DCA analysis
print('=== DETAILED DCA STRATEGY ANALYSIS ===')
print()

# Track position building
pos = 0.0
cost = 0.0
opens_in_cycle = []
cycle_num = 0
all_cycles = []

for f in data:
    sz = float(f['sz'])
    px = float(f['px'])
    dir_ = f['dir']
    
    if dir_ == 'Open Long':
        pos += sz
        cost += sz * px
        opens_in_cycle.append((sz, px))
    elif dir_ == 'Close Long':
        if pos > 0:
            cycle_num += 1
            avg_entry = cost / pos
            exit_px = px
            profit_pct = (exit_px - avg_entry) / avg_entry * 100
            profit_usd = pos * (exit_px - avg_entry)
            
            all_cycles.append({
                'cycle': cycle_num,
                'num_opens': len(opens_in_cycle),
                'total_size': pos,
                'avg_entry': avg_entry,
                'exit_px': exit_px,
                'profit_pct': profit_pct,
                'profit_usd': profit_usd,
                'sizes': [s for s, p in opens_in_cycle],
                'prices': [p for s, p in opens_in_cycle]
            })
            
            pos = 0
            cost = 0
            opens_in_cycle = []
