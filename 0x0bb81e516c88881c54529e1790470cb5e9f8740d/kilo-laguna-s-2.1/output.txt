# Analyze the buy-sell cycles

print("=== TRADING CYCLES (Buy -> Close pattern) ===")
cycles = []
current_cycle = {'buys': [], 'sells': [], 'start_time': None, 'end_time': None}
total_pnl = 0.0

for f in data_sorted:
    pnl = float(f['closedPnl'])
    if f['side'] == 'B' and 'Open Long' in f['dir']:
        if not current_cycle['start_time']:
            current_cycle['start_time'] = f['time']
        current_cycle['buys'].append(f)
    elif f['side'] == 'A' and 'Close Long' in f['dir']:

        current_cycle['end_time'] = f['time']
